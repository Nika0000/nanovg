cmake_minimum_required(VERSION 3.15)
project(nanovg_vulkan_starter LANGUAGES C)

include(FetchContent)
find_package(Vulkan REQUIRED)
find_package(glfw3 QUIET)
if(NOT TARGET glfw)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(glfw)
endif()

add_library(nanovg STATIC extern/nanovg/src/nanovg.c)
target_include_directories(nanovg PUBLIC extern/nanovg/include/nanovg extern/nanovg/third_party)
add_executable(nanovg_vulkan src/main.c)
target_include_directories(nanovg_vulkan PRIVATE src extern/nanovg/shaders/vulkan)
target_compile_definitions(nanovg_vulkan PRIVATE NANOVG_VULKAN)
target_link_libraries(nanovg_vulkan PRIVATE nanovg glfw Vulkan::Vulkan)
if(UNIX)
    target_link_libraries(nanovg_vulkan PRIVATE m)
endif()
add_custom_command(TARGET nanovg_vulkan POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:nanovg_vulkan>/assets")
