cmake_minimum_required(VERSION 3.15)
project(nanovg_webgpu_starter LANGUAGES C)

include(FetchContent)
find_package(glfw3 QUIET)
if(NOT TARGET glfw)
    set(GLFW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(GLFW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(GLFW_INSTALL OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(glfw GIT_REPOSITORY https://github.com/glfw/glfw.git GIT_TAG 3.4 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(glfw)
endif()

if(NOT NANOVG_WEBGPU_TARGET)
    find_package(Dawn QUIET)
    if(TARGET dawn::webgpu_dawn)
        set(NANOVG_WEBGPU_TARGET dawn::webgpu_dawn)
    endif()
endif()
if(NOT NANOVG_WEBGPU_TARGET)
    message(FATAL_ERROR "Set NANOVG_WEBGPU_TARGET to a Dawn or wgpu-native CMake target.")
endif()

add_library(nanovg STATIC extern/nanovg/src/nanovg.c)
target_include_directories(nanovg PUBLIC extern/nanovg/include/nanovg extern/nanovg/third_party)
add_executable(nanovg_webgpu src/main.c)
target_include_directories(nanovg_webgpu PRIVATE src extern/nanovg/shaders/wgpu)
target_compile_definitions(nanovg_webgpu PRIVATE NANOVG_WEBGPU)
target_link_libraries(nanovg_webgpu PRIVATE nanovg glfw ${NANOVG_WEBGPU_TARGET})
if(UNIX)
    target_link_libraries(nanovg_webgpu PRIVATE m)
endif()
add_custom_command(TARGET nanovg_webgpu POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:nanovg_webgpu>/assets")
